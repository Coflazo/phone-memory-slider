cmake_minimum_required(VERSION 3.24)

project(PhoneMemorySlider VERSION 1.0.0 LANGUAGES C CXX)

set(CPACK_PACKAGE_NAME "PhoneMemorySlider")
set(CPACK_PACKAGE_VENDOR "Phone Memory Slider contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Local-only phone gallery review")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "PhoneMemorySlider-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(PMS_BUILD_TESTS "Build Phone Memory Slider tests" ON)
option(PMS_BUILD_DESKTOP "Build the Qt desktop application" ON)
option(PMS_ENABLE_DEMO_AUTOMATION "Compile the screenshot/demo automation harness" OFF)
option(PMS_BUILD_FUZZ "Build the libFuzzer targets in fuzz/ (clang or MSVC); instruments every target" OFF)

if(PMS_BUILD_TESTS)
  include(CTest)
  enable_testing()
endif()

if(MSVC)
  add_compile_options(/W4 /WX /permissive- /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()

if(PMS_BUILD_FUZZ)
  # Coverage and sanitizers in every target, vendored code included (it resets its compile options, not these flags).
  if(MSVC)
    set(pms_fuzz_flags "/fsanitize=address /fsanitize-coverage=inline-8bit-counters /fsanitize-coverage=edge /fsanitize-coverage=trace-cmp /fsanitize-coverage=trace-div /Z7")
  else()
    set(pms_fuzz_flags "-fsanitize=fuzzer-no-link,address,undefined -fno-sanitize-recover=all -g")
    add_link_options(-fsanitize=address,undefined)
  endif()
  string(APPEND CMAKE_C_FLAGS " ${pms_fuzz_flags}")
  string(APPEND CMAKE_CXX_FLAGS " ${pms_fuzz_flags}")
endif()

add_subdirectory(third_party/noise-c)
add_subdirectory(core)
if(PMS_BUILD_FUZZ)
  add_subdirectory(fuzz)
else()
  add_subdirectory(tools)
endif()

if(PMS_BUILD_DESKTOP)
  find_package(Qt6 6.8 QUIET COMPONENTS Bluetooth Concurrent Quick QuickControls2 Multimedia Sql Test)
  if(Qt6_FOUND)
    qt_standard_project_setup(REQUIRES 6.8)
    add_subdirectory(desktop)
  else()
    message(STATUS "Qt 6.8+ not found; desktop target is disabled")
  endif()
endif()

if(PMS_BUILD_TESTS)
  add_subdirectory(core/tests)
endif()

include(CPack)
